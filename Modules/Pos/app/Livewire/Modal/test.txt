    /**
     * Handle incoming Paystack webhook events.
     *
     * @param Request $request
     * @return \Illuminate\Http\JsonResponse
     */
    public function handleWebhook(Request $request)
    {
        // Step 1: Verify the Paystack signature
        $signature = $request->header('X-Paystack-Signature');
        $payload   = $request->getContent();

        if (!$signature || $signature !== hash_hmac('sha512', $payload, $this->secretKey)) {
            return Response::json(['message' => 'Invalid signature'], 400);
        }

        // Step 2: Decode payload and log it
        $data = json_decode($payload, true);
        Log::info('Received Paystack Webhook:', $data);

        // Step 3: Handle only successful charges for now
        if ($data['event'] !== 'charge.success') {
            return Response::json(['message' => 'Unhandled event'], 400);
        }

        // Step 4: Find the transaction using the reference
        $reference   = $data['data']['reference'] ?? null;
        $transaction = $reference
            ? Transaction::where('reference', $reference)->first()
            : null;

        if (!$transaction) {
            Log::warning("Transaction with reference {$reference} not found.");
            return Response::json(['message' => 'Transaction not found'], 404);
        }

        // Step 5: Update transaction status
        $transaction->update([
            'status'       => $data['data']['status'],
            'processed_at' => now(),
        ]);

        // Step 6: Determine the source of payment (booking, pos, subscription, etc.)
        $type = $data['data']['metadata']['source'] ?? null;

        switch ($type) {
            case 'booking':
                $this->handleBooking($data['data']);
                break;

            case 'pos':
                $this->handlePOS($data['data']);
                break;

            case 'subscription':
                $this->handleSubscription($data['data']);
                break;

            default:
                Log::warning("Unhandled payment type: {$type}");
        }

        return Response::json(['message' => 'Webhook processed successfully']);
    }

    /**
     * Handle booking-related payment logic.
     */
    protected function handleBooking(array $payload): void
    {
        $bookingId = $payload['metadata']['booking_id'] ?? null;

        if ($bookingId) {
            // Example logic: mark booking as paid
            $booking = Booking::find($bookingId);
            if ($booking) {
                $booking->update([
                    'status' => 'confirmed',
                    'paid_at' => now(),
                ]);
                Log::info("Booking {$bookingId} marked as paid.");
            }
        }
    }

    /**
     * Handle POS-related payment logic.
     */
    protected function handlePOS(array $payload): void
    {
        $orderId = $payload['metadata']['order_id'] ?? null;

        if ($orderId) {
            // Example: confirm POS order
            $order = PosOrder::find($orderId);
            if ($order) {
                $order->update([
                    'status' => 'paid',
                ]);
                Log::info("POS order {$orderId} marked as paid.");
            }
        }
    }

    /**
     * Handle subscription-related payment logic.
     */
    protected function handleSubscription(array $payload): void
    {
